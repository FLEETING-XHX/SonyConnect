cmake_minimum_required(VERSION 3.24)
project(SonyConnect VERSION 0.0.3 LANGUAGES CXX RC)
get_filename_component(SONYCONNECT_WORKSPACE_ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
# Preserve existing EXE/test/report locations while the sources are classified.
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}")
set(SONYCONNECT_MSVC_DEPS_PREFIX "" CACHE STRING "Detected localized MSVC include prefix")
if(MSVC AND SONYCONNECT_MSVC_DEPS_PREFIX)
  set(CMAKE_CL_SHOWINCLUDES_PREFIX "${SONYCONNECT_MSVC_DEPS_PREFIX}")
endif()

if(NOT WIN32)
  message(FATAL_ERROR "SonyConnect desktop requires Windows 10/11 and the Windows SDK.")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
include(CTest)

add_library(sonyconnect_protocol STATIC native/src/protocol.cpp)
target_include_directories(sonyconnect_protocol PUBLIC native/include)
add_library(sonyconnect_session STATIC native/src/session.cpp)
target_link_libraries(sonyconnect_session PUBLIC sonyconnect_protocol PRIVATE ws2_32 bthprops setupapi)
add_library(sonyconnect_desktop STATIC native/src/desktop.cpp)
target_include_directories(sonyconnect_desktop PUBLIC native/include)
target_link_libraries(sonyconnect_desktop PUBLIC windowsapp ole32 uuid winhttp user32 propsys shell32 advapi32)

add_executable(SonyConnect WIN32 native/src/main.cpp native/resources/app.rc native/resources/app.manifest)
target_link_libraries(SonyConnect PRIVATE sonyconnect_session sonyconnect_desktop comctl32 shell32 advapi32 comdlg32)
target_link_libraries(SonyConnect PRIVATE windowscodecs)
target_compile_definitions(SonyConnect PRIVATE SONYCONNECT_VERSION="${PROJECT_VERSION}")
target_include_directories(SonyConnect PRIVATE native/resources)
foreach(configuration Debug Release RelWithDebInfo MinSizeRel)
  configure_file(config/update-source.json "${CMAKE_BINARY_DIR}/${configuration}/config/update-source.json" COPYONLY)
endforeach()

foreach(target sonyconnect_protocol sonyconnect_session sonyconnect_desktop SonyConnect)
  target_compile_definitions(${target} PUBLIC UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00)
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /utf-8 /permissive- /Zc:__cplusplus /Zc:preprocessor /EHsc)
  endif()
endforeach()

if(BUILD_TESTING)
  foreach(suite protocol session desktop)
    add_executable(${suite}_tests native/tests/${suite}_tests.cpp)
    target_link_libraries(${suite}_tests PRIVATE sonyconnect_${suite})
    if(MSVC)
      target_compile_options(${suite}_tests PRIVATE /W4 /utf-8 /EHsc)
    endif()
    add_test(NAME native_${suite} COMMAND ${suite}_tests)
  endforeach()
  add_test(NAME native_ui COMMAND SonyConnect --ui-self-test --report "${CMAKE_BINARY_DIR}/$<CONFIG>/ui-test-report.json")
  set_tests_properties(native_protocol native_session native_desktop native_ui PROPERTIES TIMEOUT 45)
endif()

install(TARGETS SonyConnect RUNTIME DESTINATION .)
install(DIRECTORY native/notices/ DESTINATION notices)
install(FILES config/update-source.json DESTINATION config)
install(FILES "${SONYCONNECT_WORKSPACE_ROOT}/docs/native/迁移与验证.md" DESTINATION . RENAME README-native.md)
